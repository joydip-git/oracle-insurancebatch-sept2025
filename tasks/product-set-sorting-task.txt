class HashMap<K,V> implements Map<K,V>{
	static class Node<K,V> implements Map.Entry<K,V>{
	}

	private Set<Map.Entry<K,V>> entries = new HashSet<Node<K,V>>();

	add(K key, V value){
		Node<K,V> node = new Node<K,V>(key,value);
		entries.add(node);
	}

	Set<Map.Entry<K,V>> entrySet(){
		return entries;
	}
}
